✅ 1. SwiftUI → .NET API 호출 샘플 코드
🧾 전제: .NET API에서 제공하는 URL 예시

예를 들어, 아래와 같은 API가 있다고 가정하겠습니다:

GET https://yourapi.com/api/routines


그리고 이 API는 아래와 같은 JSON 응답을 반환합니다:

[
  {
    "id": 1,
    "name": "가슴 루틴"
  },
  {
    "id": 2,
    "name": "등 루틴"
  }
]

✅ SwiftUI 모델 정의
struct Routine: Identifiable, Codable {
    var id: Int
    var name: String
}

✅ API 호출 함수
import Foundation

class RoutineViewModel: ObservableObject {
    @Published var routines: [Routine] = []

    func fetchRoutines() {
        guard let url = URL(string: "https://yourapi.com/api/routines") else { return }

        var request = URLRequest(url: url)
        request.httpMethod = "GET"
        request.setValue("application/json", forHTTPHeaderField: "Accept")

        URLSession.shared.dataTask(with: request) { data, response, error in
            if let data = data {
                do {
                    let decoded = try JSONDecoder().decode([Routine].self, from: data)
                    DispatchQueue.main.async {
                        self.routines = decoded
                    }
                } catch {
                    print("디코딩 실패: \(error)")
                }
            } else if let error = error {
                print("API 호출 실패: \(error)")
            }
        }.resume()
    }
}

✅ SwiftUI 화면 예시
import SwiftUI

struct RoutineListView: View {
    @StateObject var viewModel = RoutineViewModel()

    var body: some View {
        NavigationView {
            List(viewModel.routines) { routine in
                Text(routine.name)
            }
            .navigationTitle("루틴 목록")
            .onAppear {
                viewModel.fetchRoutines()
            }
        }
    }
}

struct Routine: Codable {
    var name: String
}

func createRoutine(name: String) {
    guard let url = URL(string: "https://yourapi.com/api/routines") else { return }

    var request = URLRequest(url: url)
    request.httpMethod = "POST"
    request.setValue("application/json", forHTTPHeaderField: "Content-Type")

    let routine = Routine(name: name)

    do {
        let jsonData = try JSONEncoder().encode(routine)
        request.httpBody = jsonData

        URLSession.shared.dataTask(with: request) { data, response, error in
            if let error = error {
                print("Error: \(error)")
            } else {
                print("루틴 저장 성공")
            }
        }.resume()

    } catch {
        print("인코딩 실패: \(error)")
    }
}

[HttpPost("login")]
public IActionResult Login([FromBody] LoginModel login)
{
    // 사용자 인증 처리 (예: DB에서 확인)

    var claims = new[] {
        new Claim(ClaimTypes.Name, login.Email)
    };

    var key = new SymmetricSecurityKey(Encoding.UTF8.GetBytes("your-secret-key"));
    var creds = new SigningCredentials(key, SecurityAlgorithms.HmacSha256);

    var token = new JwtSecurityToken(
        issuer: "yourapp",
        audience: "yourapp",
        claims: claims,
        expires: DateTime.Now.AddHours(1),
        signingCredentials: creds
    );

    return Ok(new {
        token = new JwtSecurityTokenHandler().WriteToken(token)
    });
}

📱 SwiftUI에서 JWT 토큰 로그인 처리
func login(email: String, password: String, completion: @escaping (String?) -> Void) {
    guard let url = URL(string: "https://yourapi.com/api/login") else { return }

    var request = URLRequest(url: url)
    request.httpMethod = "POST"
    request.setValue("application/json", forHTTPHeaderField: "Content-Type")

    let credentials = ["email": email, "password": password]
    let json = try! JSONSerialization.data(withJSONObject: credentials)

    request.httpBody = json

    URLSession.shared.dataTask(with: request) { data, response, error in
        guard let data = data,
              let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any],
              let token = json["token"] as? String else {
            completion(nil)
            return
        }
        completion(token)
    }.resume()
}

Mac 1대로만 개발을 하려는 상황에서는, 백엔드 서버와 데이터베이스까지 Mac 로컬에서 모두 구동하여 SwiftUI 앱과 API 연동 테스트까지 가능한 환경을 구성할 수 있습니다.

✅ 목표: Mac 1대로 SwiftUI + 백엔드(API) + DB까지 전체 개발
🔧 전체 구성도
[SwiftUI 앱 (Xcode)]
        ↓ API 요청 (localhost 또는 127.0.0.1)
[로컬 백엔드 서버 (.NET API)]
        ↓ DB 연결
[로컬 DB (SQL Server, PostgreSQL 등)]

✅ 1. SwiftUI 앱 개발 (Mac + Xcode)

Xcode에서 SwiftUI 프로젝트 생성

URLSession을 통해 로컬 API 호출 (http://127.0.0.1:5000/api/xxx)

앱은 시뮬레이터 또는 실제 아이폰에서 실행 가능

API 호출은 백엔드 서버가 Mac에서 실행 중일 때만 작동

✅ 2. .NET 백엔드 서버 (Mac에서 실행 가능)
🧰 사전 설치

[.NET SDK 설치 (macOS)]
👉 https://dotnet.microsoft.com/en-us/download

확인:

dotnet --version

📦 프로젝트 생성
dotnet new webapi -n MyApi
cd MyApi
dotnet run


실행 후: http://localhost:5000/swagger 또는 https://localhost:5001/swagger

💡 SwiftUI에서 localhost 호출 시 주의:

시뮬레이터에서는 localhost 사용 가능

http://127.0.0.1:5000

실제 iPhone 디바이스에서는 안됨

로컬 서버를 외부 접속 가능하게 설정하거나, ngrok 사용 필요

✅ 3. DB 구성 (Mac 로컬에서 실행)
🧩 선택 가능한 DB
DB	설치 방법	장점
PostgreSQL	Homebrew로 간단 설치	안정성, 무료, 인기
SQLite	별도 설치 필요 없음	파일 기반, 간단 테스트
SQL Server	Docker로 가능	무거움, .NET과 잘 맞음
✅ PostgreSQL 설치 (Mac)
brew install postgresql
brew services start postgresql


접속 확인:

psql postgres


DB 생성:

CREATE DATABASE fitness_db;

✅ .NET 프로젝트에서 DB 연결

NuGet 패키지 설치

dotnet add package Npgsql.EntityFrameworkCore.PostgreSQL


appsettings.json 수정:

{
  "ConnectionStrings": {
    "DefaultConnection": "Host=localhost;Database=fitness_db;Username=postgres;Password=yourpass"
  }
}


EF Core 마이그레이션 설정:

dotnet ef migrations add InitialCreate
dotnet ef database update


⚠️ EF CLI가 없을 경우:

dotnet tool install --global dotnet-ef
